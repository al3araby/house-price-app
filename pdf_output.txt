def parse_amount(x): 
if not isinstance(x, str): 
return None 
x = x.strip().lower() 
try: 
if "lac" in x: 
return float(x.replace("lac", "").strip()) * 1e5 
if "cr" in x: 
return float(x.replace("cr", "").strip()) * 1e7 
return float(x.replace(",", "")) 
except ValueError: 
return None
